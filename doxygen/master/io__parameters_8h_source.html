32inline void
33add_equal_spacing_outputs(unsigned int n_outputs,
34 unsigned int n_increments,
35 std::set<unsigned int> &output_list)
36{
37 output_list.clear();
38 if (n_increments == 0)
39 {
40 output_list.insert(0);
41 return;
42 }
43 if (!n_outputs)
44 {
45 return;
46 }
47
48 unsigned int period = std::max(1U, n_increments / n_outputs);
49 for (unsigned int output = 0; output <= n_increments; output += period)
50 {
51 output_list.insert(output);
52 }
53}

55inline void
56add_log_spacing_outputs(unsigned int n_outputs,
57 unsigned int n_increments,
58 std::set<unsigned int> &output_list)
59{
60 output_list.clear();
61 if (n_increments == 0)
62 {
63 output_list.insert(0);
64 return;
65 }
66 if (!n_outputs)
67 {
68 return;
69 }
70
71 for (unsigned int output = 1; output <= n_outputs; output++)
72 {
73 output_list.insert((unsigned int) (std::round(
74 std::pow(double(n_increments), double(output) / double(n_outputs)))));
75 }
76}

78inline void
79add_n_per_decade_outputs(unsigned int n_outputs,
80 unsigned int n_increments,
81 std::set<unsigned int> &output_list)
82{
83 output_list.clear();
84 if (n_increments == 0)
85 {
86 output_list.insert(0);
87 return;
88 }
89 if (!n_outputs)
90 {
91 return;
92 }
93
94 output_list.insert(0);
95 output_list.insert(1);
96 for (unsigned int iteration = 2; iteration <= n_increments; iteration++)
97 {
98 const auto decade = (unsigned int) (std::ceil(std::log10(iteration)));
99 const auto step_size = (unsigned int) (std::pow(10, decade) / n_outputs);
100 if (iteration % step_size == 0)
101 {
102 output_list.insert(iteration);
103 }
104 }
105}
