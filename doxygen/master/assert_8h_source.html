45[[noreturn]] inline void
46failure_handler(const libassert::assertion_info &info)
47{
48 libassert::enable_virtual_terminal_processing_if_needed();
49
50 // Create a message with and without terminal codes
51 std::string message_no_terminal_codes =
52 info.to_string(libassert::terminal_width(libassert::stderr_fileno),
53 libassert::color_scheme::blank);
54 std::string message =
55 info.to_string(libassert::terminal_width(libassert::stderr_fileno),
56 libassert::isatty(libassert::stderr_fileno)
57 ? libassert::get_color_scheme()
58 : libassert::color_scheme::blank);
59
60 // Print the message to cerr as well as the log file
61 Logger::instance() << LogFormatter::verbose(message_no_terminal_codes);
62 std::cerr << message << std::endl;
63
64 switch (info.type)
65 {
66 case libassert::assert_type::debug_assertion:
67 throw std::runtime_error(message);
68 case libassert::assert_type::assertion:
69 case libassert::assert_type::assumption:
70 case libassert::assert_type::panic:
71 case libassert::assert_type::unreachable:
72 std::fflush(stderr);
73 std::abort();
74 default:
75 std::cerr << "Critical error: Unknown libassert::assert_type" << std::endl;
76 std::abort();
77 }
78}
