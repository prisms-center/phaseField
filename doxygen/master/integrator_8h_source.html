33 template <int rank>
34 static dealii::Vector<number>
35 integrate(const dealii::DoFHandler<dim> &dof_handler, const auto &solution_vector)
36 {
37 constexpr unsigned int expected_components =
38 dealii::Tensor<rank, dim>::n_independent_components;
39 const unsigned int n_components = dof_handler.get_fe().n_components();
40
41 DEBUG_ASSERT(n_components == expected_components,
42 "The provided DoFHandler does not have the same number of components as "
43 "the expected ones. For scalar fields there should be 1 component.");
44
45 solution_vector.update_ghost_values();
46
47 static dealii::FEValues<dim> fe_values(dof_handler.get_fe(),
48 SystemWide<dim, degree>::quadrature,
49 dealii::update_values |
50 dealii::update_JxW_values);
51
52 const unsigned int num_quad_points = SystemWide<dim, degree>::quadrature.size();
53
54 std::vector<dealii::Vector<number>> quad_values(num_quad_points,
55 dealii::Vector<number>(
56 expected_components));
57
58 dealii::Vector<number> value(expected_components);
59 for (const auto &cell : dof_handler.active_cell_iterators())
60 {
61 if (cell->is_locally_owned())
62 {
63 fe_values.reinit(cell);
64 fe_values.get_function_values(solution_vector, quad_values);
65
66 // Sum up the product of the JxW and values at each quadrature point to
67 // compute the element integral.
68 for (unsigned int q_point = 0; q_point < num_quad_points; ++q_point)
69 {
70 for (unsigned int component = 0; component < expected_components;
71 component++)
72 {
73 value[component] +=
74 quad_values[q_point][component] * fe_values.JxW(q_point);
75 }
76 }
77 }
78 }
79
80 for (unsigned int component = 0; component < expected_components; component++)
81 {
82 value[component] = dealii::Utilities::MPI::sum(value[component], MPI_COMM_WORLD);
83 }
84 return value;
85 }

90 template <typename Function, typename... SolutionVectors>
91 static number
92 integrate_function(const dealii::DoFHandler<dim> &dof_handler,
93 Function &&function,
94 const SolutionVectors &...solution_vectors)
95 {
96 // compile time check
97 static_assert(sizeof...(SolutionVectors) > 0,
98 "At least one solution vector must be supplied.");
99
100 DEBUG_ASSERT(dof_handler.get_fe().n_components() == 1,
101 "The input must be a scalar field.");
102
103 (solution_vectors.update_ghost_values(), ...);
104
105 const auto &quadrature = SystemWide<dim, degree>::quadrature;
106
107 static dealii::FEValues<dim> fe_values(SystemWide<dim, degree>::fe_systems[Scalar],
108 quadrature,
109 dealii::update_values |
110 dealii::update_JxW_values);
111
112 const unsigned int num_quad_points = quadrature.size();
113
114 const auto solutions = std::forward_as_tuple(solution_vectors...);
115
116 auto quad_values = std::make_tuple(
117 (static_cast<void>(solution_vectors), std::vector<number>(num_quad_points))...);
118
119 number value {};
120
121 for (const auto &cell : dof_handler.active_cell_iterators())
122 {
123 if (cell->is_locally_owned())
124 {
125 fe_values.reinit(cell);
126
127 [&]<std::size_t... indices>(std::index_sequence<indices...>)
128 {
129 (fe_values.get_function_values(std::get<indices>(solutions),
130 std::get<indices>(quad_values)),
131 ...);
132 }(std::index_sequence_for<SolutionVectors...> {});
133
134 for (unsigned int q = 0; q < num_quad_points; ++q)
135 {
136 const number fvalue = std::apply(
137 [&](const auto &...values) -> number
138 {
139 return std::invoke(function, values[q]...);
140 },
141 quad_values);
142
143 value += fvalue * fe_values.JxW(q);
144 }
145 }
146 }
147
148 return dealii::Utilities::MPI::sum(value, MPI_COMM_WORLD);
149 }

163 solution_vectors...);
164 }
