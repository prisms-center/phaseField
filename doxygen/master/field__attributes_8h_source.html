80inline std::map<std::string, Types::Index>
81field_index_map(const std::vector<FieldAttributes> &fields)
82{
83 std::map<std::string, Types::Index> map;
84 for (unsigned int i = 0; i < fields.size(); ++i)
85 {
86 ASSERT(!map.contains(fields[i].name),
87 "The names of the fields must be unique",
88 fields[i].name);
89 map[fields[i].name] = i;
90 }
91 return map;
92}
