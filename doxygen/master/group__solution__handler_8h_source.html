339template <unsigned int dim, typename number>
340template <unsigned int degree>
341inline void
342GroupSolutionHandler<dim, number>::reinit_mg_transfer(
343 const DoFManager<dim, degree> &dof_manager)
344{
345 // 1. Initialize constraints
346 const unsigned int num_blocks = solve_block.field_indices.size();
347 mg_constraints = std::vector<dealii::MGConstrainedDoFs>(num_blocks);
348 mg_transfer = std::vector<dealii::MGTransferMatrixFree<dim, number>>(num_blocks);
349 for (unsigned int block_index = 0; block_index < num_blocks; block_index++)
350 {
351 unsigned int field_index = block_to_global_index[block_index];
352 mg_constraints[block_index].initialize(
353 dof_manager.get_field_dof_handler(field_index));
354 // 2. Initialize MG Transfer
355 mg_transfer[block_index].initialize_constraints(mg_constraints[block_index]);
356 mg_transfer[block_index].build(dof_manager.get_field_dof_handler(field_index));
357 }
358}

360template <unsigned int dim, typename number>
361template <unsigned int degree>
362inline void
363GroupSolutionHandler<dim, number>::mg_transfer_down(
364 const DoFManager<dim, degree> &dof_manager,
365 unsigned int finest_level,
366 bool transfer_old_solutions)
367{
368 const unsigned int num_blocks = solve_block.field_indices.size();
369 for (unsigned int block_index = 0; block_index < num_blocks; block_index++)
370 {
371 unsigned int field_index = block_to_global_index[block_index];
372 dealii::MGLevelObject<SolutionVector<number>> temp_mg_solutions(
373 1 + finest_level - solution_levels.size(),
374 finest_level);
375
376 const auto &dof_handler = dof_manager.get_field_dof_handler(field_index);
377
378 // transfer regular solutions to mg levels
379 mg_transfer[block_index].interpolate_to_mg(dof_handler,
380 temp_mg_solutions,
381 primary_solutions.solutions.block(
382 block_index));
383 // swap to actual mg solution vectors
384 for (unsigned int relative_level = 0; relative_level < solution_levels.size();
385 ++relative_level)
386 {
387 unsigned int level = finest_level - relative_level;
388 solution_levels[relative_level]
389 .solutions.block(block_index)
390 .swap(temp_mg_solutions[level]);
391 solution_levels[relative_level].solutions.update_ghost_values();
392 }
393
394 if (!transfer_old_solutions)
395 {
396 return;
397 }
398 // Transfer old solutions
399 for (unsigned int age_index = 0;
400 age_index < solution_levels[0].old_solutions.size();
401 ++age_index)
402 {
403 mg_transfer[block_index].interpolate_to_mg(
404 dof_handler,
405 temp_mg_solutions,
406 solution_levels[0].old_solutions[age_index].block(block_index));
407 for (unsigned int relative_level = 0; relative_level < solution_levels.size();
408 ++relative_level)
409 {
410 unsigned int level = finest_level - relative_level;
411 if (age_index < solution_levels[relative_level].old_solutions.size())
412 {
413 solution_levels[relative_level]
414 .old_solutions[age_index]
415 .block(block_index)
416 .swap(temp_mg_solutions[level]);
417 solution_levels[relative_level]
418 .old_solutions[age_index]
419 .update_ghost_values();
420 }
421 }
422 }
423 }
424}
