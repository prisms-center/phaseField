50 int_type
51 overflow(int_type int_char) override
52 {
53 if (traits_type::eq_int_type(int_char, traits_type::eof()))
54 {
55 return traits_type::not_eof(int_char);
56 }
57 if (!active)
58 {
59 return int_char;
60 }
61 bool is_ok = true;
62 const char _char = traits_type::to_char_type(int_char);
63 for (auto *buffer : buffers)
64 {
65 if (traits_type::eq_int_type(buffer->sputc(_char), traits_type::eof()))
66 {
67 is_ok = false;
68 }
69 }
70 return is_ok ? int_char : traits_type::eof();
71 }
72
73 std::streamsize
74 xsputn(const char *_char, std::streamsize n) override
75 {
76 if (!active)
77 {
78 return n;
79 }
80 std::streamsize min_written = n;
81 for (auto *buffer : buffers)
82 {
83 const std::streamsize written = buffer->sputn(_char, n);
84 min_written = std::min(min_written, written);
85 }
86 return min_written;
87 }
88
89 int
90 sync() override
91 {
92 if (!active)
93 {
94 return 0;
95 }
96 bool is_ok = true;
97 for (auto *buffer : buffers)
98 {
99 if (buffer->pubsync() != 0)
100 {
101 is_ok = false;
102 }
103 }
104 return is_ok ? 0 : -1;
105 }
